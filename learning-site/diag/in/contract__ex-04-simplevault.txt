pragma solidity ^0.8.20;

/// @title SimpleVault：ERC-4626 风格的最小收益金库
/// @notice 教学示例，未经审计，不可用于生产。
/// @dev 配套章节：Contract/07-defi-primitives.md
///      不完整实现 ERC-4626 标准，只演示"份额记账"和"汇率增长"模型。
///
///      汇率增长模型：
///      - 用户存入资产（assets），得到份额（shares）。份额数量之后不再变化。
///      - 收益（利息、手续费）直接进入金库，totalAssets 变大，totalSupply 不变。
///      - 所以"每份额对应的资产"上涨。用户取款时按新汇率换回更多资产。
///
///      首存通胀攻击风险：totalAssets 读的是 balanceOf，任何人可以直接转币"捐赠"。
///      金库几乎为空时，攻击者可以抬高汇率，让后来者的份额向下取整为 0。
///      详见 ../../Contract-Security/05-arithmetic-precision.md 第 4 节。
///      本合约的缓解：虚拟份额偏移（10^DECIMALS_OFFSET 个虚拟份额 + 1 个虚拟资产）。

interface IERC20 {
    function transfer(address to, uint256 amount) external returns (bool);
    function transferFrom(address from, address to, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
}

contract SimpleVault {
    IERC20 public immutable asset;

    uint256 public totalSupply;                 // 份额总量
    mapping(address => uint256) public balanceOf; // 每个地址的份额

    // 虚拟份额偏移：攻击者要捐赠约 10^3 倍资金，才能造成同样的取整损失
    uint256 private constant VIRTUAL_SHARES = 10 ** 3;
    uint256 private constant VIRTUAL_ASSETS = 1;

    error ZeroShares();
    error TransferFailed();

    event Deposit(address indexed sender, address indexed receiver, uint256 assets, uint256 shares);
    event Withdraw(address indexed owner, address indexed receiver, uint256 assets, uint256 shares);

    constructor(IERC20 _asset) {
        asset = _asset;
    }

    /// @notice 金库管理的资产总量 = 金库地址上的底层代币余额
    function totalAssets() public view returns (uint256) {
        return asset.balanceOf(address(this));
    }

    /// @notice 资产 → 份额（向下取整，对金库有利）
    function convertToShares(uint256 assets) public view returns (uint256) {
        return (assets * (totalSupply + VIRTUAL_SHARES)) / (totalAssets() + VIRTUAL_ASSETS);
    }

    /// @notice 份额 → 资产（向下取整，对金库有利）
    function convertToAssets(uint256 shares) public view returns (uint256) {
        return (shares * (totalAssets() + VIRTUAL_ASSETS)) / (totalSupply + VIRTUAL_SHARES);
    }

    /// @notice 存入 assets，给 receiver 铸造份额
    function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
        shares = convertToShares(assets); // 必须在转账之前计算，否则 totalAssets 已包含本次存款
        if (shares == 0) revert ZeroShares(); // 防止"存了钱却拿到 0 份额"

        if (!asset.transferFrom(msg.sender, address(this), assets)) revert TransferFailed();
        totalSupply += shares;
        balanceOf[receiver] += shares;
        emit Deposit(msg.sender, receiver, assets, shares);
    }

    /// @notice 取出 assets 给 receiver，从调用者处销毁份额
    function withdraw(uint256 assets, address receiver) external returns (uint256 shares) {
        // 取款时份额向上取整：宁可多扣用户一点份额，也不让金库吃亏
        uint256 num = assets * (totalSupply + VIRTUAL_SHARES);
        uint256 den = totalAssets() + VIRTUAL_ASSETS;
        shares = (num + den - 1) / den;
        if (shares == 0) revert ZeroShares();

        balanceOf[msg.sender] -= shares; // 份额不足时自动回滚
        totalSupply -= shares;
        emit Withdraw(msg.sender, receiver, assets, shares);

        if (!asset.transfer(receiver, assets)) revert TransferFailed(); // 先改账，后转账（CEI）
    }
}
