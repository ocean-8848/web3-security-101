pragma solidity ^0.8.20;

/// @title 01-Escrow：买方-卖方-仲裁人 ETH 托管合约
/// @notice 教学示例，未经审计，不可用于生产。
/// @dev 配套章节：Contract/01-what-is-smart-contract.md
///      状态机：Created → Funded → Released / Refunded；Funded → Disputed → Released / Refunded
contract Escrow {
    enum State { Created, Funded, Disputed, Released, Refunded }

    address public immutable buyer;   // 买方：部署者，负责存入货款
    address public immutable seller;  // 卖方：收货款的一方
    address public immutable arbiter; // 仲裁人：只在争议状态下裁决
    uint256 public amount;            // 托管金额（wei）
    State public state;               // 当前状态，初始值为 Created

    event Funded(address indexed buyer, uint256 amount);
    event Disputed(address indexed by);
    event Released(address indexed seller, uint256 amount);
    event Refunded(address indexed buyer, uint256 amount);

    error WrongState(State expected, State actual);
    error NotAuthorized(address caller);
    error ZeroAmount();
    error TransferFailed();

    /// 检查当前状态。状态不对时整笔交易回滚。
    modifier inState(State expected) {
        if (state != expected) revert WrongState(expected, state);
        _;
    }

    constructor(address _seller, address _arbiter) {
        buyer = msg.sender;
        seller = _seller;
        arbiter = _arbiter;
        // state 默认等于 Created（枚举第一个值）
    }

    /// 转移 1：Created → Funded。买方存入 ETH。
    function deposit() external payable inState(State.Created) {
        if (msg.sender != buyer) revert NotAuthorized(msg.sender); // Checks
        if (msg.value == 0) revert ZeroAmount();
        amount = msg.value;                                         // Effects
        state = State.Funded;
        emit Funded(msg.sender, msg.value);                         // 没有外部调用
    }

    /// 转移 2：Funded → Released。买方确认收货，货款付给卖方。
    function confirmDelivery() external inState(State.Funded) {
        if (msg.sender != buyer) revert NotAuthorized(msg.sender);
        _release();
    }

    /// 转移 3：Funded → Refunded。卖方主动退款给买方。
    function refundBySeller() external inState(State.Funded) {
        if (msg.sender != seller) revert NotAuthorized(msg.sender);
        _refund();
    }

    /// 转移 4：Funded → Disputed。买方或卖方发起争议，资金冻结待裁决。
    function raiseDispute() external inState(State.Funded) {
        if (msg.sender != buyer && msg.sender != seller) revert NotAuthorized(msg.sender);
        state = State.Disputed;
        emit Disputed(msg.sender);
    }

    /// 转移 5：Disputed → Released 或 Refunded。仲裁人裁决。
    function resolve(bool payToSeller) external inState(State.Disputed) {
        if (msg.sender != arbiter) revert NotAuthorized(msg.sender);
        if (payToSeller) _release();
        else _refund();
    }

    /// Checks-Effects-Interactions：调用方已完成 Checks。
    /// 这里先改状态（Effects），最后才转账（Interactions）。
    /// 如果先转账再改状态，卖方合约可以在收款回调里重入，重复领取。
    function _release() private {
        uint256 value = amount;
        amount = 0;                                  // Effects
        state = State.Released;
        emit Released(seller, value);
        (bool ok, ) = seller.call{value: value}(""); // Interactions
        if (!ok) revert TransferFailed();
    }

    function _refund() private {
        uint256 value = amount;
        amount = 0;                                  // Effects
        state = State.Refunded;
        emit Refunded(buyer, value);
        (bool ok, ) = buyer.call{value: value}("");  // Interactions
        if (!ok) revert TransferFailed();
    }
}
