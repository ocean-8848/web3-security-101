pragma solidity ^0.8.20;

/// @title BondingCurve：虚拟储备常数乘积联合曲线
/// @notice 教学示例，仿 Pump.fun 虚拟储备联合曲线，未经审计，不可用于生产。
/// @dev 配套章节：Contract/apps/Y06-pumpfun.md
///      Pump.fun 在 Solana 上用 SOL 计价；本示例改用 ETH 计价，只演示定价原理。
///      定价公式：(virtualEth) × (virtualToken) = k。
///      "虚拟"的意思：初始储备里有一部分 ETH 并不存在，只用来给出起始价格。
///      代币只记在内部 mapping 里，不是 ERC-20，不能转账。
contract BondingCurve {
    // ---------- 曲线参数（部署时固定） ----------
    uint256 public immutable initialVirtualEth;   // 初始虚拟 ETH，例如 30 ether
    uint256 public immutable graduationThreshold; // 真实 ETH 储备达到该值即毕业，例如 85 ether
    uint256 public constant FEE_BPS = 100;        // 手续费 1% = 100 / 10000
    uint256 private constant BPS = 10_000;

    // ---------- 曲线状态 ----------
    uint256 public virtualEth;       // 当前 ETH 侧储备（虚拟 + 真实）
    uint256 public virtualToken;     // 当前代币侧储备（虚拟 + 尚未卖出的真实代币）
    uint256 public realTokenReserve; // 曲线上还能卖出的真实代币数量
    bool public graduated;           // 毕业后禁止在曲线上交易

    address public immutable feeRecipient; // 平台收费地址
    uint256 public feesAccrued;            // 已累计、未提取的手续费
    mapping(address => uint256) public balanceOf; // 内部代币余额

    event Buy(address indexed buyer, uint256 ethIn, uint256 fee, uint256 tokensOut);
    event Sell(address indexed seller, uint256 tokensIn, uint256 ethOut, uint256 fee);
    event Graduated(uint256 realEth, uint256 tokensLeftForDex);

    error AlreadyGraduated();
    error ZeroAmount();
    error Slippage(uint256 out, uint256 minOut);
    error ExceedsCurve();
    error InsufficientBalance();
    error TransferFailed();

    /// @param _virtualEth   初始虚拟 ETH（示例 30 ether）
    /// @param _virtualToken 初始虚拟代币（示例 1_073_000_000e18）
    /// @param _realToken    曲线可卖出的真实代币（示例 793_100_000e18）
    /// @param _threshold    毕业阈值，以真实 ETH 储备计（示例 85 ether）
    constructor(uint256 _virtualEth, uint256 _virtualToken, uint256 _realToken, uint256 _threshold) {
        require(_realToken < _virtualToken, "real must < virtual");
        initialVirtualEth = _virtualEth;
        virtualEth = _virtualEth;
        virtualToken = _virtualToken;
        realTokenReserve = _realToken;
        graduationThreshold = _threshold;
        feeRecipient = msg.sender;
    }

    /// @notice 真实 ETH 储备 = 当前 ETH 侧储备 − 初始虚拟 ETH（不含手续费）
    function realEthReserve() public view returns (uint256) {
        return virtualEth - initialVirtualEth;
    }

    /// @notice 当前边际价格（每个代币值多少 wei，放大 1e18）
    function spotPrice() external view returns (uint256) {
        return (virtualEth * 1e18) / virtualToken;
    }

    // ---------- 报价：只读，不改状态 ----------

    /// @notice 付 ethIn（含手续费）能买到多少代币
    /// 公式：tokensOut = virtualToken × ethNet / (virtualEth + ethNet)
    function quoteBuy(uint256 ethIn) public view returns (uint256 tokensOut, uint256 fee) {
        fee = (ethIn * FEE_BPS) / BPS;
        uint256 ethNet = ethIn - fee;
        tokensOut = (virtualToken * ethNet) / (virtualEth + ethNet);
    }

    /// @notice 卖 tokensIn 个代币能拿回多少 ETH（已扣手续费）
    /// 公式：ethGross = virtualEth × tokensIn / (virtualToken + tokensIn)
    function quoteSell(uint256 tokensIn) public view returns (uint256 ethOut, uint256 fee) {
        uint256 ethGross = (virtualEth * tokensIn) / (virtualToken + tokensIn);
        fee = (ethGross * FEE_BPS) / BPS;
        ethOut = ethGross - fee;
    }

    // ---------- 交易 ----------

    /// @notice 用 ETH 买代币。minOut 是滑点保护：到手少于它就回滚。
    function buy(uint256 minOut) external payable returns (uint256 tokensOut) {
        if (graduated) revert AlreadyGraduated();
        if (msg.value == 0) revert ZeroAmount();
        uint256 fee;
        (tokensOut, fee) = quoteBuy(msg.value);
        if (tokensOut < minOut) revert Slippage(tokensOut, minOut);
        // 不能卖出超过曲线上剩余的真实代币。真实实现会退回多付的部分，这里直接回滚。
        if (tokensOut > realTokenReserve) revert ExceedsCurve();

        // 先改状态（检查-生效-交互顺序）
        virtualEth += msg.value - fee;
        virtualToken -= tokensOut;
        realTokenReserve -= tokensOut;
        feesAccrued += fee;
        balanceOf[msg.sender] += tokensOut;
        emit Buy(msg.sender, msg.value, fee, tokensOut);

        _checkGraduation();
    }

    /// @notice 把代币卖回曲线。minEthOut 是滑点保护。
    function sell(uint256 tokensIn, uint256 minEthOut) external returns (uint256 ethOut) {
        if (graduated) revert AlreadyGraduated();
        if (tokensIn == 0) revert ZeroAmount();
        if (balanceOf[msg.sender] < tokensIn) revert InsufficientBalance();
        uint256 fee;
        (ethOut, fee) = quoteSell(tokensIn);
        if (ethOut < minEthOut) revert Slippage(ethOut, minEthOut);

        // 先改状态，再转 ETH，防止重入
        balanceOf[msg.sender] -= tokensIn;
        virtualToken += tokensIn;
        realTokenReserve += tokensIn;
        virtualEth -= ethOut + fee;
        feesAccrued += fee;
        emit Sell(msg.sender, tokensIn, ethOut, fee);

        (bool ok, ) = msg.sender.call{value: ethOut}("");
        if (!ok) revert TransferFailed();
    }

    /// @notice 平台提取手续费
    function withdrawFees() external {
        uint256 amount = feesAccrued;
        feesAccrued = 0;
        (bool ok, ) = feeRecipient.call{value: amount}("");
        if (!ok) revert TransferFailed();
    }

    /// @dev 真实 ETH 储备达到阈值即毕业。
    ///      真实平台在这一步把真实 ETH 和预留代币（示例 206_900_000e18）注入 DEX 池，
    ///      并销毁或锁定 LP 凭证，让流动性无法被撤走。
    ///      Pump.fun 早期迁移到 Raydium，2025-03 起迁移到自有 PumpSwap。
    ///      本示例只发事件，不实现迁移。
    function _checkGraduation() private {
        if (realEthReserve() >= graduationThreshold) {
            graduated = true;
            emit Graduated(realEthReserve(), realTokenReserve);
        }
    }
}
